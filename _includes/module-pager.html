{%- comment -%} Previous / next navigation for any module page, rendered at the very bottom of the page (called from related-pages.html). The order comes from the page's sidebar data file: only top-level sidebar items whose title starts with a two-digit number ("00 ...", "11 ...") are part of the sequence, so unnumbered items ("All resources") sit outside it. The count uses those numbers: "00" is shown as "Before you start" and the highest number is the total. Pages outside the sequence get a "Back to section" link when either: - they are nested under a numbered item in the sidebar (found automatically), or - they set `back_to: ` in front matter. A module with an empty sidebar simply renders nothing. {%- endcomment -%} {%- assign pager_nav = site.data.sidebars[page.sidebar] -%} {%- assign pager_items = "" | split: "" -%} {%- assign pager_digits = "0123456789" | split: "" -%} {%- for item in pager_nav.subitems -%} {%- assign pager_c1 = item.title | slice: 0 -%} {%- assign pager_c2 = item.title | slice: 1 -%} {%- if pager_digits contains pager_c1 and pager_digits contains pager_c2 -%} {%- assign pager_items = pager_items | push: item -%} {%- endif -%} {%- endfor -%} {%- assign pager_url = page.url | remove: ".html" -%} {%- assign pager_index = nil -%} {%- assign pager_parent_item = nil -%} {%- for item in pager_items -%} {%- if item.url == pager_url -%}{%- assign pager_index = forloop.index0 -%}{%- endif -%} {%- if pager_parent_item == nil -%} {%- for sub in item.subitems -%} {%- if sub.url == pager_url -%}{%- assign pager_parent_item = item -%}{%- endif -%} {%- endfor -%} {%- endif -%} {%- endfor -%} {%- assign pager_overview = pager_nav.title_url | default: pager_nav["title-url"] -%} {%- assign pager_overview = pager_overview | prepend: "/" | append: "/" | replace: "//", "/" -%} {%- if pager_index != nil -%} {%- assign pager_count = pager_items.size -%} {%- assign pager_last = pager_items | last -%} {%- assign pager_total = pager_last.title | slice: 0, 2 | plus: 0 -%} {%- assign pager_current = pager_items[pager_index] -%} {%- assign pager_position = pager_current.title | slice: 0, 2 | plus: 0 -%} {%- assign pager_percent = pager_position | times: 100 | divided_by: pager_total -%} {%- assign pager_prev_index = pager_index | minus: 1 -%} {%- assign pager_next_index = pager_index | plus: 1 -%}
